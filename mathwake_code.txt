package com.example.mathwake.alarm

import android.content.BroadcastReceiver
import android.content.Context
import android.content.Intent
import android.os.Build

class AlarmReceiver : BroadcastReceiver() {
    override fun onReceive(context: Context, intent: Intent) {
        if (intent.action == Intent.ACTION_BOOT_COMPLETED) {
            // Need to reschedule all active alarms
            return
        }

        val alarmId = intent.getIntExtra("ALARM_ID", -1)
        val difficulty = intent.getStringExtra("DIFFICULTY") ?: "EASY"
        
        if (alarmId != -1) {
            val serviceIntent = Intent(context, AlarmService::class.java).apply {
                putExtra("ALARM_ID", alarmId)
                putExtra("DIFFICULTY", difficulty)
            }
            
            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
                context.startForegroundService(serviceIntent)
            } else {
                context.startService(serviceIntent)
            }
        }
    }
}

package com.example.mathwake.alarm

import android.app.AlarmManager
import android.app.PendingIntent
import android.content.Context
import android.content.Intent
import com.example.mathwake.data.Alarm
import java.util.Calendar

class AlarmScheduler(private val context: Context) {

    private val alarmManager = context.getSystemService(Context.ALARM_SERVICE) as AlarmManager

    fun schedule(alarm: Alarm) {
        if (!alarm.isActive) return

        val intent = Intent(context, AlarmReceiver::class.java).apply {
            putExtra("ALARM_ID", alarm.id)
            putExtra("DIFFICULTY", alarm.difficulty.name)
        }

        val pendingIntent = PendingIntent.getBroadcast(
            context,
            alarm.id,
            intent,
            PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE
        )

        val calendar = Calendar.getInstance().apply {
            set(Calendar.HOUR_OF_DAY, alarm.hour)
            set(Calendar.MINUTE, alarm.minute)
            set(Calendar.SECOND, 0)
            
            // If the time is in the past, schedule for tomorrow
            if (before(Calendar.getInstance())) {
                add(Calendar.DATE, 1)
            }
        }

        try {
            alarmManager.setExactAndAllowWhileIdle(
                AlarmManager.RTC_WAKEUP,
                calendar.timeInMillis,
                pendingIntent
            )
        } catch (e: SecurityException) {
            // Need to handle missing EXACT_ALARM permission
            e.printStackTrace()
        }
    }

    fun cancel(alarm: Alarm) {
        val intent = Intent(context, AlarmReceiver::class.java)
        val pendingIntent = PendingIntent.getBroadcast(
            context,
            alarm.id,
            intent,
            PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE
        )
        alarmManager.cancel(pendingIntent)
    }
}

package com.example.mathwake.alarm

import android.app.NotificationChannel
import android.app.NotificationManager
import android.app.PendingIntent
import android.app.Service
import android.content.Context
import android.content.Intent
import android.media.RingtoneManager
import android.os.IBinder
import androidx.core.app.NotificationCompat
import com.example.mathwake.MainActivity

class AlarmService : Service() {

    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        val alarmId = intent?.getIntExtra("ALARM_ID", -1) ?: -1
        val difficulty = intent?.getStringExtra("DIFFICULTY") ?: "EASY"

        createNotificationChannel()

        val fullScreenIntent = Intent(this, MainActivity::class.java).apply {
            addFlags(Intent.FLAG_ACTIVITY_NEW_TASK or Intent.FLAG_ACTIVITY_CLEAR_TASK)
            putExtra("IS_RINGING", true)
            putExtra("ALARM_ID", alarmId)
            putExtra("DIFFICULTY", difficulty)
        }

        val fullScreenPendingIntent = PendingIntent.getActivity(
            this,
            0,
            fullScreenIntent,
            PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE
        )

        val notification = NotificationCompat.Builder(this, CHANNEL_ID)
            .setContentTitle("MathWake Alarm")
            .setContentText("Time to wake up!")
            .setSmallIcon(android.R.drawable.ic_lock_idle_alarm)
            .setPriority(NotificationCompat.PRIORITY_HIGH)
            .setCategory(NotificationCompat.CATEGORY_ALARM)
            .setFullScreenIntent(fullScreenPendingIntent, true)
            .setOngoing(true)
            .build()

        startForeground(1, notification)

        // For ringing sound, we could use MediaPlayer with a loop
        // In a real app, we handle the MediaPlayer instance here
        playAlarmSound()

        return START_STICKY
    }

    private fun playAlarmSound() {
        val alarmUri = RingtoneManager.getDefaultUri(RingtoneManager.TYPE_ALARM)
        val ringtone = RingtoneManager.getRingtone(this, alarmUri)
        ringtone?.play()
        // Note: For a robust alarm, a foreground service managing a MediaPlayer on loop is better.
        // We'll keep it simple with RingtoneManager for now.
    }

    override fun onBind(intent: Intent?): IBinder? {
        return null
    }

    private fun createNotificationChannel() {
        val channel = NotificationChannel(
            CHANNEL_ID,
            "Alarm Channel",
            NotificationManager.IMPORTANCE_HIGH
        ).apply {
            description = "Channel for Alarm notifications"
        }
        val notificationManager = getSystemService(Context.NOTIFICATION_SERVICE) as NotificationManager
        notificationManager.createNotificationChannel(channel)
    }

    companion object {
        private const val CHANNEL_ID = "MATHWAKE_ALARM_CHANNEL"
    }
}

package com.example.mathwake.logic

import com.example.mathwake.data.MathDifficulty
import kotlin.random.Random

data class MathProblem(
    val question: String,
    val answer: Int
)

object MathProblemGenerator {
    fun generateProblem(difficulty: MathDifficulty): MathProblem {
        return when (difficulty) {
            MathDifficulty.EASY -> generateEasy()
            MathDifficulty.MEDIUM -> generateMedium()
            MathDifficulty.HARD -> generateHard()
        }
    }

    private fun generateEasy(): MathProblem {
        // 1-digit addition or subtraction (positive answer)
        val isAddition = Random.nextBoolean()
        val a = Random.nextInt(1, 10)
        val b = Random.nextInt(1, 10)
        
        return if (isAddition) {
            MathProblem("$a + $b = ?", a + b)
        } else {
            val max = maxOf(a, b)
            val min = minOf(a, b)
            MathProblem("$max - $min = ?", max - min)
        }
    }

    private fun generateMedium(): MathProblem {
        // 2-digit addition or subtraction
        val isAddition = Random.nextBoolean()
        val a = Random.nextInt(10, 100)
        val b = Random.nextInt(10, 100)

        return if (isAddition) {
            MathProblem("$a + $b = ?", a + b)
        } else {
            val max = maxOf(a, b)
            val min = minOf(a, b)
            MathProblem("$max - $min = ?", max - min)
        }
    }

    private fun generateHard(): MathProblem {
        // Mixed: 2-digit multiplication, division (whole numbers) or complex addition
        val op = Random.nextInt(0, 3)
        return when (op) {
            0 -> { // Multiplication 1-digit by 2-digit
                val a = Random.nextInt(2, 10)
                val b = Random.nextInt(11, 30)
                MathProblem("$a × $b = ?", a * b)
            }
            1 -> { // Division (ensure whole number)
                val b = Random.nextInt(2, 15)
                val answer = Random.nextInt(5, 20)
                val a = answer * b
                MathProblem("$a ÷ $b = ?", answer)
            }
            else -> { // Complex addition 3 items
                val a = Random.nextInt(10, 50)
                val b = Random.nextInt(10, 50)
                val c = Random.nextInt(10, 50)
                MathProblem("$a + $b + $c = ?", a + b + c)
            }
        }
    }
}

@Composable
fun AlarmRingingScreen(
    alarmId: Int,
    difficulty: MathDifficulty,
    onAlarmDismissed: () -> Unit
) {
    // Prevent accidental back press during alarm ringing
    BackHandler {
        // Do nothing, force user to solve math
    }

    var problem by remember { mutableStateOf<MathProblem?>(null) }
    // rememberSaveable to keep state when rotating or switching apps
    var currentAnswer by rememberSaveable { mutableStateOf("") }
    var isError by rememberSaveable { mutableStateOf(false) }

    LaunchedEffect(difficulty) {
        if (problem == null) {
            problem = MathProblemGenerator.generateProblem(difficulty)
        }
    }

    Box(modifier = Modifier.fillMaxSize(), contentAlignment = Alignment.Center) {
        Column(horizontalAlignment = Alignment.CenterHorizontally) {
            Text(stringResource(R.string.wake_up_message))
            
            problem?.let { p ->
                Text(p.question)
                // In a real app, this would be a custom Numpad. 
                // For simplicity in structure, just a button to simulate correct answer
                Text("Current input: $currentAnswer")

                if (isError) {
                    Text(stringResource(R.string.incorrect_answer), color = androidx.compose.ui.graphics.Color.Red)
                }

                Button(onClick = { 
                    if (currentAnswer == p.answer.toString()) {
                        onAlarmDismissed()
                    } else {
                        // Simulate error
                        isError = true
                    }
                }) {
                    Text(stringResource(R.string.submit_answer))
                }
                
                // Cheat button for testing
                Button(onClick = { currentAnswer = p.answer.toString() }) {
                    Text("Cheat (Set correct answer)")
                }
            }
        }
    }
}

package com.example.mathwake

import android.os.Bundle
import androidx.activity.ComponentActivity
import androidx.activity.compose.setContent
import androidx.activity.enableEdgeToEdge
import androidx.compose.foundation.layout.fillMaxSize
import androidx.compose.material3.Scaffold
import androidx.compose.ui.Modifier
import androidx.navigation.NavType
import androidx.navigation.compose.NavHost
import androidx.navigation.compose.composable
import androidx.navigation.compose.rememberNavController
import androidx.navigation.navArgument
import com.example.mathwake.data.MathDifficulty
import com.example.mathwake.ui.screens.AddEditAlarmScreen
import com.example.mathwake.ui.screens.AlarmListScreen
import com.example.mathwake.ui.screens.AlarmRingingScreen
import com.example.mathwake.ui.theme.MathWakeTheme

class MainActivity : ComponentActivity() {
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        enableEdgeToEdge()
        
        val isRinging = intent.getBooleanExtra("IS_RINGING", false)
        val ringingAlarmId = intent.getIntExtra("ALARM_ID", -1)
        val ringingDifficulty = intent.getStringExtra("DIFFICULTY") ?: "EASY"
        
        setContent {
            MathWakeTheme {
                val navController = rememberNavController()
                
                Scaffold(modifier = Modifier.fillMaxSize()) { innerPadding ->
                    NavHost(
                        navController = navController, 
                        startDestination = if (isRinging) "ringing" else "list"
                    ) {
                        composable("list") {
                            AlarmListScreen(
                                onNavigateToAdd = { navController.navigate("add_edit/-1") },
                                onNavigateToEdit = { id -> navController.navigate("add_edit/$id") }
                            )
                        }
                        
                        composable(
                            "add_edit/{alarmId}",
                            arguments = listOf(navArgument("alarmId") { type = NavType.IntType })
                        ) { backStackEntry ->
                            val alarmId = backStackEntry.arguments?.getInt("alarmId")
                            AddEditAlarmScreen(
                                alarmId = if (alarmId == -1) null else alarmId,
                                onNavigateBack = { navController.popBackStack() }
                            )
                        }
                        
                        composable("ringing") {
                            AlarmRingingScreen(
                                alarmId = ringingAlarmId,
                                difficulty = MathDifficulty.valueOf(ringingDifficulty),
                                onAlarmDismissed = { finish() }
                            )
                        }
                    }
                }
            }
        }
    }
}
